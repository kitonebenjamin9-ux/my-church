rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn(){ return request.auth != null; }
    function isAdmin(){
      return signedIn() &&
        exists(/databases/$(database)/documents/users/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role in ['admin','pastor'];
    }
    function member(){ return signedIn() && !isAdmin(); }

    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid==uid || isAdmin());
      allow create: if signedIn() && request.auth.uid==uid && request.resource.data.role=='member';
      allow update: if signedIn() && request.auth.uid==uid && request.resource.data.role==resource.data.role;
      allow delete: if isAdmin();
    }

    // Public, administrator-managed content.
    match /siteSettings/{id} { allow read: if true; allow write: if isAdmin(); }
    match /branches/{id} { allow read: if true; allow write: if isAdmin(); }
    match /ministries/{id} { allow read: if true; allow write: if isAdmin(); }
    match /ministryLeaders/{id} { allow read: if true; allow write: if isAdmin(); }
    match /events/{id} { allow read: if true; allow write: if isAdmin(); }
    match /sermons/{id} { allow read: if true; allow write: if isAdmin(); }
    match /announcements/{id} { allow read: if true; allow write: if isAdmin(); }
    match /dailyVerses/{id} { allow read: if true; allow write: if isAdmin(); }
    match /gallery/{id} { allow read: if true; allow write: if isAdmin(); }
    match /heroVideos/{id} { allow read: if true; allow write: if isAdmin(); }
    match /churchProject/{id} { allow read: if true; allow write: if isAdmin(); }
    match /notifications/{id} { allow read,write: if isAdmin(); }

    match /videoTestimonies/{id} {
      allow read: if resource.data.status == 'approved' || isAdmin();
      allow create: if request.resource.data.status == 'pending';
      allow update, delete: if isAdmin();
    }

    match /contactMessages/{id} {
      allow create: if request.resource.data.status == 'new';
      allow read, update, delete: if isAdmin();
    }

    match /registrations/{id} {
      allow create: if true;
      allow read, update, delete: if isAdmin();
    }

    match /eventRegistrations/{id} {
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow read: if signedIn() && (resource.data.uid == request.auth.uid || isAdmin());
      allow update, delete: if isAdmin();
    }

    match /prayerRequests/{id} {
      allow create: if signedIn() && request.resource.data.uid == request.auth.uid;
      allow read: if signedIn() && (resource.data.uid == request.auth.uid || isAdmin());
      allow update, delete: if isAdmin();
    }
  }
}
